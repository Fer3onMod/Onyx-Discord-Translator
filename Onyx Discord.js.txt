    function protectText(text){
        if(!config.protection){
            return {
                text:text,
                restore:x=>x
            };
        }

        const tokens=[];
        const regexes=[
            /```[\s\S]*?```/g,
            /`[^`\n]+`/g,
            /https?:\/\/[^\s]+/gi,
            /www\.[^\s]+/gi,
            /<@!?\d+>/g,
            /<@&\d+>/g,
            /<#\d+>/g,
            /<a?:[A-Za-z0-9_]+:\d+>/g,
            /#[A-Za-z0-9_-]{2,}/g
        ];

        let output=text;

        regexes.forEach(regex=>{
            output=output.replace(regex,match=>{
                const token=`ONYXLOCK${tokens.length}END`;
                tokens.push(match);
                return token;
            });
        });

        return {
            text:output,
            restore:translated =>
                translated.replace(
                    /ONYXLOCK(\d+)END/g,
                    (_,n)=>tokens[Number(n)] || ""
                )
        };
    }

    function translate(text,lang){
        return new Promise(resolve=>{

            if(!text || text.trim().length<2){
                resolve({
                    text:null,
                    detected:null
                });
                return;
            }

            const key=`${text}__${lang}__${config.protection}`;

            if(state.cache.has(key)){
                resolve(state.cache.get(key));
                return;
            }

            const pack=protectText(text);

            GM_xmlhttpRequest({
                method:"GET",

                url:
                    "https://translate.googleapis.com/translate_a/single" +
                    "?client=gtx" +
                    "&sl=auto" +
                    "&tl="+encodeURIComponent(lang) +
                    "&dt=t" +
                    "&q="+encodeURIComponent(pack.text),

                onload:response=>{

                    try{

                        const data=
                            JSON.parse(
                                response.responseText
                            );

                        const raw=
                            Array.isArray(data?.[0])
                                ? data[0]
                                    .map(x=>x?.[0]||"")
                                    .join("")
                                : "";

                        const result={
                            text:raw
                                ? pack.restore(raw)
                                : null,

                            detected:
                                typeof data?.[2]==="string"
                                    ? data[2]
                                    : null
                        };

                        if(result.text){
                            state.cache.set(
                                key,
                                result
                            );
                        }

                        resolve(result);

                    }catch(e){

                        console.error(
                            "[Onyx] Translation error",
                            e
                        );

                        resolve({
                            text:null,
                            detected:null
                        });
                    }
                },

                onerror:()=>{
                    resolve({
                        text:null,
                        detected:null
                    });
                }
            });
        });
    }

    function addHistory(
        source,
        target,
        original,
        translated,
        detected
    ){

        if(
            !config.history ||
            !translated
        ){
            return;
        }

        state.history.unshift({
            id:
                Date.now()+Math.random(),

            time:
                new Date().toISOString(),

            source,
            target,
            original,
            translated,
            detected
        });

        if(
            state.history.length>100
        ){
            state.history.length=100;
        }

        save(
            "akram_history",
            state.history
        );
    }

    ){

                state.favorites.slice(0,50);

        save(
            "akram_favorites",
            state.favorites
        );
    }

    async function translateOutgoingBody(
        body
    ){

        if(
            !config.enabled ||
            typeof body!=="string" ||
            !body.trim() ||
            state.outgoing
        ){
            return body;
        }

        let data;

        try{
            data=JSON.parse(body);
        }catch{
            return body;
        }

        if(
            !data ||
            typeof data.content!=="string"
        ){
            return body;
        }

        const original=
            data.content;

        if(
            !original.trim() ||
            original.trim().startsWith("/")
        ){
            return body;
        }

        const target=
            outgoingLanguage();

        try{

            state.outgoing=true;

            setStatus(
                "Translating outgoing..."
            );

            const result=
                await translate(
                    original,
                    target
                );

            if(
                result.text &&
                result.text
                    .trim()
                    .toLowerCase() !==
                original
                    .trim()
                    .toLowerCase()
            ){

                data.content=
                    result.text;

                addHistory(
                    result.detected||"auto",
                    target,
                    original,
                    result.text,
                    result.detected||""
                );

                stat("translations");
                stat("outgoing");

                setStatus(
                    "Ready"
                );

                return JSON.stringify(
                    data
                );
            }

        }catch(e){

            console.error(
                "[Onyx] Outgoing error",
                e
            );

        }finally{

            state.outgoing=false;
        }

        setStatus("Ready");

        return body;
    }

    const nativeFetch=
        window.fetch.bind(
            window
        );

    window.fetch=
        async function(
            resource,
            options
        ){

            try{

                let url="";
                let method="";
                let body=null;

                if(
                    resource instanceof
                    Request
                ){

                    url=
                        resource.url||"";

                    method=
                        (
                            options?.method||
                            resource.method||
                            ""
                        )
                            .toUpperCase();

                    if(
                        options &&
                        Object.prototype
                            .hasOwnProperty.call(
                                options,
                                "body"
                            )
                    ){

                        body=options.body;

                    }else if(
                        method==="POST"
                    ){

                        try{
                            body=
                                await resource
                                    .clone()
                                    .text();
                        }catch{
                            body=null;
                        }
                    }

                }else{

                    url=
                        typeof resource==="string"
                            ? resource
                            : resource?.url||"";

                    method=
                        String(
                            options?.method||""
                        ).toUpperCase();

                    body=
                        options?.body??null;
                }

                const messageRequest=
                    method==="POST" &&
                    url.includes("/api/") &&
                    url.includes("/messages") &&
                    typeof body==="string";

                if(messageRequest){

                    const translated=
                        await translateOutgoingBody(
                            body
                        );

                    if(translated!==body){

                        if(
                            resource instanceof Request &&
                            (
                                !options ||
                                !Object.prototype
                                    .hasOwnProperty.call(
                                        options,
                                        "body"
                                    )
                            )
                        ){

                            const request=
                                new Request(
                                    resource.url,
                                    {
                                        method:"POST",
                                        headers:
                                            new Headers(
                                                resource.headers
                                            ),
                                        body:translated,
                                        credentials:
                                            resource.credentials,
                                        mode:
                                            resource.mode,
                                        cache:
                                            resource.cache,
                                        redirect:
                                            resource.redirect,
                                        referrer:
                                            resource.referrer,
                                        referrerPolicy:
                                            resource.referrerPolicy,
                                        integrity:
                                            resource.integrity
                                    }
                                );

                            return nativeFetch(
                                request
                            );
                        }

                        return nativeFetch(
                            resource,
                            {
                                ...(options||{}),
                                body:translated
                            }
                        );
                    }
                }

            }catch(e){

                console.error(
                    "[Onyx] fetch interceptor error",
                    e
                );
            }

            return nativeFetch(
                resource,
                options
            );
        };

    const nativeOpen=
        XMLHttpRequest.prototype.open;

    const nativeSend=
        XMLHttpRequest.prototype.send;

    XMLHttpRequest.prototype.open=
        function(
            method,
            url,
            ...rest
        ){

            this.__onyxMethod=
                String(method||"");

            this.__onyxURL=
                typeof url==="string"
                    ? url
                    : url?.href||"";

            return nativeOpen.call(
                this,
                method,
                url,
                ...rest
            );
        };

    XMLHttpRequest.prototype.send=
        function(body){

            const method=
                String(
                    this.__onyxMethod||""
                ).toUpperCase();

            const url=
                String(
                    this.__onyxURL||""
                );

            if(
                config.enabled &&
                method==="POST" &&
                url.includes("/api/") &&
                url.includes("/messages") &&
                typeof body==="string"
            ){

                const xhr=this;

                translateOutgoingBody(
                    body
                )
                .then(
                    result=>{
                        try{
                            nativeSend.call(
                                xhr,
                                result
                            );
                        }catch{
                            nativeSend.call(
                                xhr,
                                body
                            );
                        }
                    }
                )
                .catch(
                    ()=>{
                        nativeSend.call(
                            xhr,
                            body
                        );
                    }
                );

                return;
            }

            return nativeSend.call(
                this,
                body
            );
        };

    ){

                new InputEvent(

                        inputType:"insertText",

                    }
                )
            );

        }catch{

            input.dispatchEvent(
                new Event(
                    "input",
                    {
                        bubbles:true
                    }
                )
            );
        }
    }

        }

    function startDictation(){

            ()=>{

                    "mic-active"
                );
            };

            async event=>{

                const transcript=
                    event
                        .results?.[0]?.[0]
                        ?.transcript||
                    "";

                if(!transcript)
                    return;

                const input=
                    getInput();

                if(!input)
                    return;

                replaceInput(
                    input,
                    transcript
                );

                const target=
                    outgoingLanguage();

                const result=
                    await translate(
                        transcript,
                        target
                    );

                if(result.text){

                    replaceInput(
                        input,
                        result.text
                    );

                    addHistory(
                        result.detected||config.read,
                        target,
                        transcript,
                        result.text,
                        result.detected||""
                    );
                }

                stat("voice");
                stat("translations");
            };

        try{
            state.recognition.start();
        }catch{}
    }

    function speakText(text){

            !(
                "speechSynthesis"
                in window
            ) ||
            !text
        ){
            return;
        }

        speechSynthesis.cancel();

        const utterance=
            new SpeechSynthesisUtterance(
                text
            );

        utterance.lang=
            config.read;

        utterance.rate=
            Number(
                config.ttsRate
            )||.9;

        utterance.volume=
            Number(
                config.ttsVolume
            );

        speechSynthesis.speak(
            utterance
        );
    }

    function messageText(el){

        const clone=
            el.cloneNode(true);

        clone
            .querySelectorAll(
                ".tg-msg-wrapper,.time,.message-time,span.time,time"
            )

            clone.innerText||
            clone.textContent||
            ""
        ).trim();
    }

        el,
        original,
        translated,
        detected

                ".tg-msg-wrapper"
            )
        ){
            return;
        }

        el.classList.add(
            "tg-checked"
        );

            document.createElement(
                "div"
            );

        wrapper.className=
            "tg-msg-wrapper";

        const box=
            document.createElement(
                "div"
            );

        box.className=
            "tg-msg-trans";

        applyMessageStyle(box);

        const textNode=
            document.createTextNode(
                translated
            );

        box.appendChild(
            textNode
        );

        const actions=
            document.createElement(
                "div"
            );

        actions.className=

            document.createElement(

                    box.firstChild

            document.createElement(

        tts.className=
            "tg-mini-btn";

        tts.title=
            "Listen";

            I.volume;

        tts.onclick=
            e=>{

                e.stopPropagation();

                speakText(
                    box.firstChild
                        ?.textContent||
                    translated
                );
            };

        /* FAVORITE */

        const fav=
            document.createElement(
                "div"

        fav.className=
            "tg-mini-btn";

        fav.title=
            "Favorite";

        fav.innerHTML=
            I.star;

        if(

                    x.target===config.read
            )
        ){

                );

                if(!box.isConnected)

                    index<translated.length

        if(config.multi){

                                }
                            }
                        );
                    }
                );
        }

        el.appendChild(
            wrapper
        );

        rememberUser(
            el,
            detected
        );

        addHistory(
            detected||"auto",
            config.read,
            original,
            translated,
            detected||""
        );

        stat("translations");
        stat("incoming");
    }

    async function processQueue(){

        if(
            state.processing ||
            !state.queue.length
        ){
            return;
        }

        state.processing=
            true;

        while(
            state.queue.length
        ){

            const item=
                state.queue.shift();

            if(
                !item?.el?.isConnected
            ){
                continue;
            }

            const result=
                await translate(
                    item.text,
                    config.read
                );

            if(
                result.text &&
                (
                    !config.smart ||
                    result.text
                        .trim()
                        .toLowerCase() !==
                    item.text
                        .trim()
                        .toLowerCase()
                )
            ){

                createMessageTranslation(
                    item.el,
                    item.text,
                    result.text,
                    result.detected
                );

            }else{

                item.el.classList.add(
                    "tg-checked"
                );
            }
        }

        state.processing=
            false;
    }

    function scanChat(){

        if(!config.enabled)
            return;

        const nodes=
            document.querySelectorAll(
                [
                    ".text-content",
                    ".translatable-message",
                    ".message-text",
                    "div[id^='message-content-']",
                    "[class*='messageContent']"
                ].join(",")
            );

        nodes.forEach(
            el=>{

                if(
                    el.classList.contains(
                        "tg-checked"
                    ) ||
                    el.dataset.onyxPending==="1"
                ){
                    return;
                }

                const text=
                    messageText(el);

                if(!text)
                    return;

                el.dataset.onyxPending=
                    "1";

                state.queue.push({
                    el,
                    text
                });

                delete el.dataset.onyxPending;
            }
        );

        processQueue();
    }

    }

    async function onTextSelection(){

        if(!config.quickSelect)
            return;

        const selection=
            window.getSelection();

        const text=
            (selection?.toString()||"")
                .trim();

        if(
            !text ||
            text.length<2
        ){
            return;
        }

        selectionPopup();

        const popup=
            document.getElementById(
                "onyx-selection-popup"
            );

        const range=
            selection.getRangeAt(0);

        const rect=
            range.getBoundingClientRect();

        popup.style.left=
            `${Math.max(
                10,
                Math.min(
                    window.innerWidth-280,
                    rect.left
                )
            )}px`;

        popup.style.top=
            `${Math.min(
                window.innerHeight-90,
                rect.bottom+8
            )}px`;

        popup.style.display=
            "block";

        document.getElementById(
            "onyx-popup-src"
        ).textContent=
            text;

        document.getElementById(
            "onyx-popup-dst"
        ).textContent=
            "Translating...";

        const result=
            await translate(
                text,
                config.read
            );

        document.getElementById(
            "onyx-popup-dst"
        ).textContent=
            result.text||
            "Translation failed";

        if(result.text){

            addHistory(
                result.detected||"auto",
                config.read,
                text,
                result.text,
                result.detected||""
            );

            stat("selected");
            stat("translations");
        }

        clearTimeout(
            onTextSelection.timer
        );

        onTextSelection.timer=
            setTimeout(
                ()=>{
                    popup.style.display=
                        "none";
                },
                6000
            );
    }

    document.addEventListener(
        "mouseup",
        ()=>{
            setTimeout(
                onTextSelection,
                20
            );
        }
    );

    document.addEventListener(
        "mousedown",
        e=>{
            const popup=
                document.getElementById(

    async function translateClipboard(){

        try{

            const text=
                await navigator.clipboard
                    .readText();

            if(!text?.trim())
                return;

            const result=
                await translate(
                    text,
                    config.write
                );

            if(result.text){

                await navigator.clipboard
                    .writeText(
                        result.text
                    );

                addHistory(
                    result.detected||"auto",
                    config.write,
                    text,
                    result.text,
                    result.detected||""
                );

                stat("clipboard");
                stat("translations");

                setStatus(
                    "Clipboard translated"
                );

                alert(
                    "Translated text copied to clipboard."
                );
            }

        }catch(e){

            alert(
                "Clipboard permission was not available."
            );
        }
    }

    function dropdown(type,value){

        const box=
            document.createElement(
                "div"
            );

        box.className=
            "onyx-langbox";

        const label=
            document.createElement(
                "div"
            );

        label.className=
            "onyx-langlabel";

        label.textContent=
            type==="read"
                ? "Incoming"
                : "Outgoing";

        const button=
            document.createElement(
                "div"
            );

        button.className=
            "onyx-select";

        button.textContent=
            LANGUAGES[value]||
            value;

        const list=
            document.createElement(
                "div"
            );

        list.className=
            "onyx-dropdown";

        const search=
            document.createElement(
                "input"
            );

        search.className=
            "onyx-search";

        search.placeholder=
            "Search language...";

        search.onclick=
            e=>
                e.stopPropagation();

        search.oninput=
            e=>{

                const q=
                    e.target.value
                        .toLowerCase();

                list
                    .querySelectorAll(
                        ".onyx-opt"
                    )
                    .forEach(
                        opt=>{
                            opt.style.display=
                                opt.textContent
                                    .toLowerCase()
                                    .includes(q)
                                    ? "block"
                                    : "none";
                        }
                    );
            };

        list.appendChild(
            search
        );

        Object.entries(
            LANGUAGES
        ).forEach(
            ([code,name])=>{

                const opt=
                    document.createElement(
                        "div"
                    );

                opt.className=
                    "onyx-opt";

                if(code===value)
                    opt.classList.add(
                        "sel"
                    );

                opt.textContent=
                    name;

                opt.onclick=
                    e=>{

                        e.stopPropagation();

                        config[type]=
                            code;

                        sync(
                            type,
                            code
                        );

                        button.textContent=
                            name;

                        list.classList.remove(
                            "show"
                        );

                        if(
                            type==="read"
                        ){
                            refreshTranslations();
                        }
                    };

                list.appendChild(
                    opt
                );
            }
        );

        button.onclick=
            e=>{

                e.stopPropagation();

                document
                    .querySelectorAll(
                        ".onyx-dropdown"
                    )
                    .forEach(
                        x=>{
                            if(x!==list){
                                x.classList.remove(
                                    "show"
                                );
                            }
                        }
                    );

                list.classList.toggle(
                    "show"
                );

                if(
                    list.classList.contains(
                        "show"
                    )
                ){

                    search.value=
                        "";

                    search.focus();
                }
            };

        box.append(
            label,
            button,
            list
        );

        return box;
    }

    function escapeHTML(text){

        return String(
            text
        ).replace(
            /[&<>'"]/g,
            char=>({
                "&":"&amp;",
                "<":"&lt;",
                ">":"&gt;",
                "'":"&#39;",
                '"':"&quot;"
            }[char])
        );
    }

                    )
                    .value;

                const list=
                    value
                        .split(",")
                        .map(
                            x=>x.trim()
                        )
                        .filter(
                            x=>LANGUAGES[x]
                        )
                        .slice(0,5);

                if(list.length){

                    config.multiTargets=

                    sync(

                    );

                    refreshTranslations();
                }
            };

        document.getElementById(
            "onyx-reset-stats"
        ).onclick=
            ()=>{

                    translations:0,
                    incoming:0,
                    outgoing:0,
                    voice:0,
                    selected:0,
                    clipboard:0
                };

                save(
                    "akram_stats",
                    state.stats
                );

                renderAdvanced();
            };

        /* FLOATING BUTTON */

        const floating=
            document.getElementById(
                "onyx-floating-toggle"
            );

        let dragging=false;

        let startX=0;
        let startY=0;

        let initialX=0;
        let initialY=0;

        floating.onmousedown=
            e=>{

                dragging=false;

                startX=
                    e.clientX;

                startY=
                    e.clientY;

                const rect=
                    floating.getBoundingClientRect();

                initialX=
                    rect.left;

                initialY=
                    rect.top;

                const move=
                    event=>{

                        const dx=
                            event.clientX-
                            startX;

                        const dy=
                            event.clientY-
                            startY;

                        if(
                            Math.abs(dx)>3 ||
                            Math.abs(dy)>3
                        ){
                            dragging=true;
                        }

                        if(dragging){

                            floating.style.left=
                                `${initialX+dx}px`;

                            floating.style.top=
                                `${initialY+dy}px`;

                            floating.style.right=
                                "auto";

                            floating.style.bottom=
                                "auto";
                        }
                    };

                const up=
                    ()=>{
                        document.removeEventListener(
                            "mousemove",
                            move
                        );

                        document.removeEventListener(
                            "mouseup",
                            up
                        );

                        if(!dragging){
                            togglePanel();
                        }
                    };

                document.addEventListener(
                    "mousemove",
                    move
                );

                document.addEventListener(
                    "mouseup",
                    up
                );
            };
    }

    function setupInputButtons(){

        const input=
            getInput();

        if(!input)
            return;

        const target=
            input.closest(
                ".input-message-container"
            ) ||
            input.closest(
                "form"
            );

        if(!target)
            return;

        const buttons=
            target.querySelector(
                'div[class*="buttons_"]'
            ) ||
            target.querySelector(
                'div[class*="buttonContainer"]'
            );

        if(!buttons)
            return;

        if(
            !buttons.querySelector(
                ".tg-input-actions"
            )
        ){

            const wrap=
                document.createElement(
                    "div"
                );

            wrap.className=
                "tg-input-actions";

            const makeButton=
                (
                    id,
                    title,
                    icon,
                    action
                )=>{

                    const button=
                        document.createElement(
                            "button"
                        );

                    button.type=
                        "button";

                    button.id=
                        id;

                    button.title=
                        title;

                    button.className=
                        "tg-action-btn";

                    button.innerHTML=
                        icon;

                    button.onmousedown=
                        e=>
                            e.preventDefault();

                    button.onclick=
                        e=>{

                            e.preventDefault();

                            e.stopPropagation();

                            action();
                        };

                    return button;
                };

            const mic=
                makeButton(
                    "tg-mic-btn",
                    "Voice Input",
                    I.mic,
                    startDictation
                );

            const translateButton=
                makeButton(
                    "tg-write-btn",
                    "Translate",
                    I.translate,
                    handleWrite
                );

            wrap.append(
                mic,
                translateButton
            );

            buttons.appendChild(
                wrap
            );
        }

        if(
            !input.dataset.onyxObserved
        ){

            input.dataset.onyxObserved=
                "1";

            input.addEventListener(
                "input",
                ()=>{

                    if(!config.live)
                        return;

                    clearTimeout(
                        state.typingTimer
                    );

                    state.typingTimer=
                        setTimeout(
                            ()=>{
                                if(
                                    getInputText(
                                        input
                                    ).length>2
                                ){
                                    handleWrite();
                                }
                            },
                            1500
                        );
                }
            );
        }
    }

                        x.classList.remove(
                            "show"
                        )
                );
        },
        true
    );

    /* =========================================================
       BOOT
       ========================================================= */

    function boot(){

        if(!document.body)
            return;

        createPanel();

        setupInputButtons();

        scanChat();

        const observer=
            new MutationObserver(
                ()=>{

                    try{
                        setupInputButtons();
                    }catch{}

                    try{
                        scanChat();
                    }catch{}

                    if(
                        !document.getElementById(
                            "tg-master-panel"
                        )
                    ){

                        try{
                            createPanel();
                        }catch{}
                    }
                }
            );

        observer.observe(
            document.body,
            {
                childList:true,
                subtree:true
            }
        );

        setInterval(
            ()=>{
                try{
                    setupInputButtons();
                }catch{}

                try{
                    scanChat();
                }catch{}
            },
            2000
        );

        setInterval(
            ()=>{
                state.cache.clear();
            },
            600000
        );

        console.log(
            "%c[Onyx Translator] PRO v43.0 loaded",
            "color:#00DC82;font-weight:800"
        );
    }

    if(
        document.readyState==="loading"
    ){

        document.addEventListener(
            "DOMContentLoaded",
            boot,
            { once:true }
        );

    }else{

        boot();
    }

})();
